% LuaLaTeX の基本：\directlua と外部 .lua ファイル
% （ratex v0.4.7 では luacode パッケージが通らないため、Lua は別ファイルに書く）
\documentclass{article}
\usepackage{amsmath}
\directlua{dofile(kpse.find_file("lua_basic.lua", "lua"))}

\begin{document}
\section*{Lua in LaTeX}

Lua version: \directlua{tex.print(_VERSION)},
LuaTeX version: \directlua{tex.print(status.luatex_version)}.

$\sqrt{2} \approx \directlua{fmt(".10f", math.sqrt(2))}$,
\quad
$2^{64} = \directlua{fmt(".0f", 2^64)}$.

$\displaystyle\sum_{k=1}^{100} k^2 = \directlua{tex.print(sum_squares(100))}$

\bigskip
\directlua{fib_table(15)}

\bigskip
Upper: \directlua{tex.print(string.upper("hello from lua"))}
\end{document}
