\begin{luacode*}
function wallis(n)
  local v = (n % 2 == 0) and math.pi / 2 or 1
  for k = n, 2, -2 do v = v * (k - 1) / k end
  tex.print(string.format("%.6f", v))
end
\end{luacode*}
